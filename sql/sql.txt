IF DB_ID('webst4') IS NULL
BEGIN
  CREATE DATABASE webst4;
END
GO

USE webst4;
GO

-- Drop (đúng thứ tự FK)
IF OBJECT_ID('dbo.otp_tokens', 'U') IS NOT NULL DROP TABLE dbo.otp_tokens;
IF OBJECT_ID('dbo.user_roles', 'U') IS NOT NULL DROP TABLE dbo.user_roles;
IF OBJECT_ID('dbo.products', 'U') IS NOT NULL DROP TABLE dbo.products;
IF OBJECT_ID('dbo.users', 'U') IS NOT NULL DROP TABLE dbo.users;
IF OBJECT_ID('dbo.roles', 'U') IS NOT NULL DROP TABLE dbo.roles;
GO

-- Roles
CREATE TABLE dbo.roles (
  id BIGINT IDENTITY(1,1) PRIMARY KEY,
  name NVARCHAR(50) NOT NULL UNIQUE
);
GO

-- Users
CREATE TABLE dbo.users (
  id BIGINT IDENTITY(1,1) PRIMARY KEY,
  username NVARCHAR(50) NOT NULL UNIQUE,
  email NVARCHAR(120) NOT NULL UNIQUE,
  password NVARCHAR(255) NOT NULL,
  full_name NVARCHAR(150) NULL,
  images NVARCHAR(1000) NULL, -- format: url|publicId
  enabled BIT NOT NULL DEFAULT(0),
  created_at DATETIME2 NOT NULL DEFAULT(SYSUTCDATETIME()),
  updated_at DATETIME2 NULL
);
GO

-- User-Roles (Many-to-many)
CREATE TABLE dbo.user_roles (
  user_id BIGINT NOT NULL,
  role_id BIGINT NOT NULL,
  CONSTRAINT PK_user_roles PRIMARY KEY (user_id, role_id),
  CONSTRAINT FK_ur_user FOREIGN KEY (user_id) REFERENCES dbo.users(id),
  CONSTRAINT FK_ur_role FOREIGN KEY (role_id) REFERENCES dbo.roles(id)
);
GO

-- Products
CREATE TABLE dbo.products (
  id BIGINT IDENTITY(1,1) PRIMARY KEY,
  name NVARCHAR(255) NOT NULL,
  description NVARCHAR(2000) NULL,
  price DECIMAL(18,2) NOT NULL DEFAULT(0),
  quantity INT NOT NULL DEFAULT(0),
  image NVARCHAR(1000) NULL, -- format: url|publicId
  created_at DATETIME2 NOT NULL DEFAULT(SYSUTCDATETIME())
);
GO

-- OTP Tokens
CREATE TABLE dbo.otp_tokens (
  id BIGINT IDENTITY(1,1) PRIMARY KEY,
  user_id BIGINT NOT NULL,
  type NVARCHAR(30) NOT NULL,         -- REGISTER | FORGOT_PASSWORD
  otp_hash NVARCHAR(255) NOT NULL,    -- bcrypt
  attempts INT NOT NULL DEFAULT(0),
  expires_at DATETIME2 NOT NULL,
  verified_at DATETIME2 NULL,
  created_at DATETIME2 NOT NULL DEFAULT(SYSUTCDATETIME()),
  CONSTRAINT FK_otp_user FOREIGN KEY (user_id) REFERENCES dbo.users(id)
);
GO

CREATE INDEX IX_otp_user_type_created ON dbo.otp_tokens(user_id, type, created_at DESC);
GO
